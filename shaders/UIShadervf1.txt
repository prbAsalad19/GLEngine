#version 460 core

in vec2 fragUV;
in vec4 fragColor;

out vec4 screenColor;

uniform sampler2D uTexture;
uniform int       uIsText;   // 0 = solid quad, 1 = text glyph

void main()
{
    if (uIsText != 0)
    {
        // RGB from vertex color (text tint), alpha from font atlas coverage.
        float coverage = texture(uTexture, fragUV).a;
	screenColor = fragColor * vec4(1.0, 1.0, 1.0, coverage);
    }
    else
    {
        screenColor = fragColor;
    }
}
